Portfolio Selection: CVXPY Model

import cvxpy as cp
 
# 1. Decision variables
S = cp.Variable()
B = cp.Variable()
I = cp.Variable()
M = cp.Variable()
 
# 2. Objective function
objective = cp.Maximize(0.1*S + 0.03*B + 0.04*I + 0.01*M)
 
# 3. Constraints
constraints = [
    S + B + I + M == 1,
    80*S + 20*B + 30*I <= 40,
    S <= 0.75,
    I - B >= 0,
    M >= 0.1,
    M <= 0.3,
    S >= 0, B >= 0, I >= 0, M >= 0,
]
